import hashlib
import json
import re
import subprocess
from datetime import datetime
from pathlib import Path
from typing import Any

import acceptance
import lifecycle

AREA = Path(__file__).resolve().parent
ROOT = Path.cwd()
MATRIX = "tests/scenarios/2026-11-28-issue-12-opencode-v2-acceptance.json"
TASK_ONE = "57cd2badd4f49a6100ba042efd89b13567790205"
NATIVE_FIXTURE = "tests/scenarios/2026-11-01-issue-24-opencode-native-fixture.json"
SETUP_IDS = set(acceptance.SETUP_CASES)
LIFECYCLE_IDS = set(lifecycle.CASES)
CONFORMANCE_IDS = {
    "skill-structure",
    "native-issue-intake",
    "language-neutral-verification",
    "lifecycle-transitions",
    "superpowers-installation",
}
ISSUE_IDS = {

    "ready",
    "selection",
    "start",

PRIVATE = re.compile(
    r"github_pat_|gh[pousr]_[A-Za-z0-9]{25,}|\bsk-[A-Za-z0-9_-]{25,}"
    r"|Authorization: (?:Basic|Bearer)|[A-Za-z]:[\\/]Users[\\/]",
    re.IGNORECASE,
)


def digest(value: Any) -> str:
    return hashlib.sha256(json.dumps(value, sort_keys=True).encode("utf-8")).hexdigest()


def recorded(source: Path) -> Any:
    source = source.resolve(strict=True)
    if not source.is_relative_to(AREA):
        raise ValueError(f"Replay receipt escaped its fixture: {source.name}")
    return json.loads(source.read_text(encoding="utf-8"))


def capture_hash(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def short_text(text: Any) -> str | None:
    if not isinstance(text, str):
        return None
    if text.startswith('{"type":"error"'):
        error = json.loads(text)["error"]
        return f"{error['type']}: {error['message']}"
    text = text.replace(str(AREA), ".sp-bp/work/issue-12")
    text = text.replace(AREA.as_posix(), ".sp-bp/work/issue-12")
    if PRIVATE.search(text):
        raise ValueError("Replay description needs manual screening")
    return text


def setup_replay(source: Path) -> dict[str, Any]:
    row = recorded(source)
    if not row.get("checks"):
        return {"result": "UNKNOWN", "observed_error": short_text(row.get("reason"))}
    score = acceptance.classify(row)
    if score != row["result"]:
        raise ValueError(f"Archived setup scorer differs for {row['id']}")
    if score == "PASS" and not acceptance.scoped_native_observation(row):
        raise ValueError(f"Scoped native setup evidence is incomplete for {row['id']}")
    results = [
        {key: item[key] for key in ("id", "status", "git_blob", "capture_sha256") if key in item}
        for item in row["native_results"]
    ]
    observations = row["observations"]
    native_before = observations["installed_before"]
    native_after = observations["installed_after"]
    package = {item["path"]: item["actual"] for item in row["package_files"]}
    exact_manifest = bool(package) and package == lifecycle.expected_package_files(
        row["candidate_sha"], ROOT
    )
    if row["expected_status"] == "completed" and score == "PASS" and not exact_manifest:
        raise ValueError(f"Positive setup package manifest differs for {row['id']}")
    return {
        "result": score,
        "host_version": row["host_version"],
        "host_sha256": row["executable_hash"],
        "backplane_revision": row["candidate_sha"],
        "upstream_origin": row["upstream_origin"],
        "upstream_revision": row["upstream_revision"],
        "checks": row["checks"],
        "model_gh_checks": observations["model_gh_checks"],
        "direct_gh_exits": [item["exit"] for item in observations["direct_gh_checks"]],
        "gh_failure_fixture_version": {
            "F-gh-intake": "2.40.1",
            "F-gh-label": "1.0.0",
            "F-gh-close": "1.14.0",
        }.get(row["id"]),
        "installed_skill_results": results,
        "diagnostic": short_text(row["native_results"][0].get("diagnostic"))
        if row["native_results"]
        else None,
        "recovery": short_text(row["recovery"]),
        "package_files": len(package),
        "exact_package_manifest": exact_manifest,
        "installed_before_sha256": digest(native_before),
        "installed_after_sha256": digest(native_after),
        "protected_before_sha256": digest(row["before"]),
        "protected_after_sha256": digest(row["after"]),
        "capture_sha256": row["capture_sha256"],
        "scored_receipt_sha256": capture_hash(source),
    }


def observed_lifecycle(item: dict[str, Any]) -> dict[str, Any]:
    observation = {
        key: item[key]
        for key in (
            "revision",
            "upstream_revision",
            "config_sha256",
            "loaded_skills",
            "load_exit",
            "discovery_prompt_sha256",
        )
        if key in item
    }
    observation.update(
        protected_sha256=digest(item["protected"]),
        skill_blobs={name: value["blob"] for name, value in item.get("skill_files", {}).items()},
        native_result_status={
            result["id"]: result["status"] for result in item.get("native_results", ())
        },
    )
    if "installed_manifest" in item:
        installed = item["installed_manifest"]
        expected = (
            lifecycle.expected_package_files(item["revision"], ROOT) if item["revision"] else {}
        )
        if installed != expected:
            raise ValueError("Native lifecycle package manifest differs from its Git tree")
        expected_skills = {lifecycle.UPSTREAM_SKILL: lifecycle.UPSTREAM_SKILL_BLOB}
        if item["revision"]:
            expected_skills.update(
                (name, expected[f"skills/{name}/SKILL.md"]) for name in lifecycle.BACKPLANE_SKILLS
            )
        if observation["skill_blobs"] != expected_skills:
            raise ValueError("Native lifecycle attributed Skill blob differs from its source")
        observation.update(package_files=len(installed), package_manifest_sha256=digest(installed))
    if "diagnostic" in item:
        observation["diagnostic"] = short_text(item["diagnostic"])
    return observation


def maintenance_checks(row: dict[str, Any], checks: dict[str, Any]) -> dict[str, Any]:
    prefix = {
        "L-update": "update",
        "L-rollback": "rollback",
        "L-unavailable-source": "unavailable",
        "L-failed-verification-recovery": "recovery",
    }.get(row["id"])
    if prefix is None:
        return {}
    revision = row["after"]["revision"]
    required = {f"{prefix}_before": revision, f"{prefix}_after": revision}
    if prefix == "recovery":
        required["denied_candidate_preflight"] = row["failure"]["revision"]
    selected = {}
    for name, expected in required.items():
        observation = checks.get(name, {})
        if (
            observation.get("resolved_revision") != expected
            or observation.get("auth_exit") != 0
            or observation.get("label_flags") is not True
            or observation.get("completed_reason") is not True
            or not observation.get("gh_version")
        ):
            raise ValueError(f"Missing or incomplete maintenance check {name}")
        selected[name] = observation
    return selected


def lifecycle_replays(candidate: str) -> dict[str, dict[str, Any]]:
    path = AREA / "native-03/lifecycle-receipt.json"
    rows = recorded(path)["rows"]
    checks = recorded(AREA / "native-03/capability-checks.json")
    install_path = AREA / "native-02/lifecycle-receipt.json"
    recovery_path = AREA / "native-04/lifecycle-receipt.json"
    recovery_rows = {row["id"]: row for row in recorded(recovery_path)["rows"]}
    checks.update(recorded(AREA / "native-04/capability-checks.json"))
    install = recorded(install_path)["rows"][0]
    rows = [
        install if item["id"] == install["id"] else recovery_rows.get(item["id"], item)
        for item in rows
    ]
    prompt = lifecycle.discovery_request()
    for row in rows:
        revision = row["after"]["revision"]
        manifest = lifecycle.expected_package_files(revision, ROOT) if revision else {}
        if not lifecycle.matches_archived_evidence(row, prompt, manifest, candidate):
            raise ValueError(f"Archived lifecycle evidence mismatch for {row['id']}")
    return {
        row["id"]: {
            "result": lifecycle.classify(row),
            "host_version": row["host_version"],
            "host_sha256": row["host_sha256"],
            "upstream_origin": "https://github.com/obra/superpowers.git",
            "before": observed_lifecycle(row["before"]),
            "after": observed_lifecycle(row["after"]),
            "failure": observed_lifecycle(row["failure"]) if "failure" in row else None,
            "checks": maintenance_checks(row, checks),
            "scored_receipt_sha256": capture_hash(
                install_path
                if row["id"] == "L-install"
                else recovery_path
                if row["id"] in recovery_rows
                else path
            ),
        }
        for row in rows
    }


def verify_authority(matrix: dict[str, Any], fixture: dict[str, Any]) -> None:
    plan = matrix.get("plan", {})
    if (
        plan.get("operator_response") != "approved"
        or not plan.get("source")
        or not plan.get("path")
        or capture_hash(ROOT / plan["path"]) != plan.get("sha256")
    ):
        raise ValueError("Exact plan approval is missing or mismatched")
    initial = fixture.get("approved_initial_issue", {})
    if initial.get("number") != 39 or not lifecycle.sha(initial.get("body_sha256"), 64):
        raise ValueError("Approved initial fixture authority is missing")
    published = fixture["fixture_publication"]
    readback = matrix["fixture_39"]
    close = next(item for item in fixture["cases"] if item["stage"] == "close")
    after = close["after"]
    if (
        published["pull_request"] != readback["pr"]
        or published["checks"]["headSha"] != published["delivery"]["revision"]
        or published["checks"]["conclusion"] != "success"
        or not (
            datetime.fromisoformat(readback["merged_at"])
            < datetime.fromisoformat(readback["closed_at"])
            <= datetime.fromisoformat(after["updatedAt"])
        )
        or after["state"] != "CLOSED"
        or after["stateReason"] != "COMPLETED"
        or [label["name"] for label in after["labels"]] != ["documentation"]
    ):
        raise ValueError("The fixture completed without verified prior integration")


def verify_fixture_stages(fixture: dict[str, Any]) -> None:
    cases = {item["stage"]: item for item in fixture["cases"]}
    if (
        len(fixture["cases"]) != len(ISSUE_IDS)
        or set(cases) != ISSUE_IDS
        or any(not item["passed"] or not item["protected_equal"] for item in cases.values())
    ):
        raise ValueError("One or more authenticated fixture stages are unverified")
    if any(cases[name]["before"] != cases[name]["after"] for name in ("status", "selection")):
        raise ValueError("Read-only fixture stage mutated state")
    for stage in cases.values():
        for state in (stage["before"], stage["after"]):
            labels = [label["name"] for label in state["labels"]]
            if state["state"] == "OPEN" and (
                "documentation" not in labels
                or sum(label.startswith("backplane:") for label in labels) != 1
            ):
                raise ValueError("An open fixture state did not preserve its labels")


def verify_retained(matrix: dict[str, Any]) -> None:
    fixture = json.loads((ROOT / NATIVE_FIXTURE).read_text(encoding="utf-8"))
    verify_authority(matrix, fixture)
    verify_fixture_stages(fixture)
    directory = ROOT / "tests/scenarios"
    original_setup = directory / "2026-11-01-issue-23-opencode-acceptance.json"
    if not acceptance.verify(original_setup)["valid"]:
        raise ValueError("The owning setup or below-floor receipt no longer verifies")
    hosted = json.loads(
        (directory / "2026-11-27-issue-23-integrated-delivery.json").read_text(encoding="utf-8")
    )["native_windows_v2_2_1_4"]
    if hosted["result"] != "PASS" or not all(hosted["checks"].values()):
        raise ValueError("Exact hosted native setup receipt did not pass")
    conformance = json.loads(
        (directory / "2026-11-01-issue-24-opencode-conformance.json").read_text(encoding="utf-8")
    )
    scores = {"skill-structure": conformance["skill_structure"], **conformance["scenarios"]}
    if (
        set(scores) != CONFORMANCE_IDS
        or not conformance["protected_equal"]
        or any(item["result"] != "PASS" for item in scores.values())
    ):
        raise ValueError("Owning conformance evidence no longer verifies")


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("output", type=Path)
    parser.add_argument("--setup", nargs="*", type=Path, default=[])
    parser.add_argument("--candidate", required=True)
    parser.add_argument("--preflight", type=Path, required=True)
    args = parser.parse_args()
    original = subprocess.check_output(
        ["git", "show", f"{TASK_ONE}:{MATRIX}"], cwd=ROOT, encoding="utf-8"
    )
    matrix = json.loads(original)
    ids = [row["id"] for row in matrix["rows"]]
    if len(ids) != 41 or set(ids) != SETUP_IDS | LIFECYCLE_IDS | CONFORMANCE_IDS | ISSUE_IDS:
        raise SystemExit("Task 1 row IDs differ from the approved matrix")
    verify_retained(matrix)
    replays = lifecycle_replays(matrix["baseline"]["revision"])
    seen = set()
    for source in args.setup:
        case = recorded(source)["id"]
        if case not in SETUP_IDS or case in seen:
            raise SystemExit(f"Invalid or duplicate setup replay: {case}")
        replays[case] = setup_replay(source)
        seen.add(case)
    for row in matrix["rows"]:
        if row["id"] in replays:
            replay = replays[row["id"]]
            row.update(result=replay["result"], disposition="replayed", replay=replay)
    preflight = recorded(args.preflight)
    if (
        preflight["http_status"] != 422
        or preflight["exit"] != 1
        or not preflight["full_issue_intake_equal"]
        or preflight["configuration_before_sha256"] != preflight["configuration_after_sha256"]
    ):
        raise SystemExit("The before-apply preflight does not establish preservation")
    manifest = lifecycle.expected_package_files(args.candidate, ROOT)
    totals = {
        result: sum(row["result"] == result for row in matrix["rows"])
        for result in ("PASS", "FAIL", "UNKNOWN")
    }
    matrix.update(
        phase="Task 2 candidate verification",
        replayed_baseline=matrix["baseline"]["revision"],
        before_apply_preflight=preflight,
        expected_candidate_package={
            "revision": args.candidate,
            "files": len(manifest),
            "manifest_sha256": digest(manifest),
            "manifest": manifest,
        },
        totals=totals,
        candidate_local_rows_pass=totals["PASS"] == 41,
        exact_hosted_candidate_result="UNKNOWN",
    )
    text = json.dumps(matrix, indent=2, ensure_ascii=False) + "\n"
    if PRIVATE.search(text):
        raise SystemExit("Output still contains an unscreened local or credential marker")
    with args.output.open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(text)
    print(json.dumps(totals))


if __name__ == "__main__":
    main()
