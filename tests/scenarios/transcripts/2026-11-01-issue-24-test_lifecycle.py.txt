"""Negative and positive controls for native lifecycle evidence scoring."""

import copy
import unittest

from lifecycle import classify

PRIOR = "a" * 40
CANDIDATE = "b" * 40
UPSTREAM = "c" * 40
FILE = "d" * 40
SKILLS = ["using-superpowers", "managing-superpowers-backlog", "managing-superpowers-handoffs"]


def recorded_row() -> dict:
    """Build the fully observed native rollback-shaped control."""
    return {
        "id": "L-rollback",
        "kind": "native",
        "host_version": "2.1.4",
        "host_sha256": "e" * 64,
        "expected_revision": PRIOR,
        "before": {
            "revision": CANDIDATE,
            "protected": {"files": {"user-work.txt": FILE}, "issue": {"number": 24}},
            "upstream_revision": UPSTREAM,
        },
        "after": {
            "revision": PRIOR,
            "protected": {"files": {"user-work.txt": FILE}, "issue": {"number": 24}},
            "upstream_revision": UPSTREAM,
            "inventory_ready": True,
            "effective_entries": 1,
            "expected_manifest": {"index.js": FILE},
            "installed_manifest": {"index.js": FILE},
            "loaded_skills": SKILLS,
            "load_exit": 0,
            "session_id": "fresh-session",
        },
    }


class LifecycleEvidenceTests(unittest.TestCase):
    """Check that mandatory evidence cannot be replaced by a nominal pass."""

    def test_complete_native_rollback(self):
        """A verified restore establishes the operation's positive result."""
        self.assertEqual(classify(recorded_row()), "PASS")

    def test_stale_inventory(self):
        """A pre-activation inventory must never establish a revision."""
        row = recorded_row()
        row["after"]["inventory_ready"] = False
        self.assertEqual(classify(row), "UNKNOWN")

    def test_wrong_restored_revision(self):
        """A ready inventory at the candidate revision is the wrong rollback."""
        row = recorded_row()
        row["after"]["revision"] = CANDIDATE
        self.assertEqual(classify(row), "FAIL")

    def test_missing_positive_skill_result(self):
        """An incomplete native load log cannot show all three skills."""
        row = recorded_row()
        row["after"]["loaded_skills"] = SKILLS[:2]
        self.assertEqual(classify(row), "UNKNOWN")

    def test_changed_protected_file(self):
        """An actual preservation failure is visible despite an incomplete capture."""
        row = recorded_row()
        row["after"]["protected"] = copy.deepcopy(row["after"]["protected"])
        row["after"]["protected"]["files"]["user-work.txt"] = "f" * 40
        row["after"]["loaded_skills"] = []
        self.assertEqual(classify(row), "FAIL")

    def test_synthetic_result(self):
        """A synthetic control cannot pass a mandatory native row."""
        row = recorded_row()
        row["kind"] = "synthetic"
        self.assertEqual(classify(row), "UNKNOWN")

    def test_available_manifest_cannot_hide_wrong_existing_revision(self):
        """An update to the existing revision is not the pinned update."""
        row = recorded_row()
        row["id"] = "L-update"
        row["before"]["revision"] = PRIOR
        self.assertEqual(classify(row), "FAIL")

    def test_failed_candidate_requires_exact_prior_configuration(self):
        """Recovery includes the prior configuration bytes as well as source."""
        row = recorded_row()
        row["id"] = "L-failed-verification-recovery"
        row["before"]["revision"] = PRIOR
        row["before"]["config_sha256"] = "1" * 64
        row["after"]["config_sha256"] = "2" * 64
        row["failure"] = {
            "result": "observed",
            "diagnostic": "A native Skill was denied",
            "protected": row["before"]["protected"],
        }
        self.assertEqual(classify(row), "FAIL")
